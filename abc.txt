WEEK 1: OVERVIEW OF C
Q1. Implement the string operations strcat(), strcmp(), and strcpy() using pointers.
Answer
#include <stdio.h>

void mystrcpy(char *d, char *s) {
    while ((*d++ = *s++) != '\0');
}

void mystrcat(char *d, char *s) {
    while (*d) d++;
    while ((*d++ = *s++) != '\0');
}

int mystrcmp(char *a, char *b) {
    while (*a && *a == *b) {
        a++;
        b++;
    }
    return *a - *b;
}

int main() {
    char s1[100] = "Hello ";
    char s2[] = "World";
    char s3[100];

    mystrcpy(s3, s2);
    printf("Copied: %s\n", s3);

    mystrcat(s1, s2);
    printf("Concatenated: %s\n", s1);

    printf("Comparison: %d\n",
           mystrcmp("Apple", "Apple"));

    return 0;
}

Output
Copied: World
Concatenated: Hello World
Comparison: 0

Q3. Implement length in feet and inches using a structure. Write routines to add, multiply, and subtract lengths.
Answer
#include <stdio.h>

struct Length {
    int feet, inches;
};

struct Length convert(int total) {
    struct Length l;

    l.feet = total / 12;
    l.inches = total % 12;

    return l;
}

struct Length add(struct Length a, struct Length b) {
    return convert((a.feet + b.feet) * 12
                   + a.inches + b.inches);
}

struct Length subtract(struct Length a, struct Length b) {
    return convert((a.feet - b.feet) * 12
                   + a.inches - b.inches);
}

struct Length multiply(struct Length a, int n) {
    return convert((a.feet * 12 + a.inches) * n);
}

void display(struct Length a) {
    printf("%d feet %d inches\n", a.feet, a.inches);
}

int main() {
    struct Length a = {5, 8}, b = {3, 9};

    printf("First length: ");
    display(a);

    printf("Second length: ");
    display(b);

    printf("Addition: ");
    display(add(a, b));

    printf("Subtraction: ");
    display(subtract(a, b));

    printf("Multiplication by 2: ");
    display(multiply(a, 2));

    return 0;
}

Output
First length: 5 feet 8 inches
Second length: 3 feet 9 inches
Addition: 9 feet 5 inches
Subtraction: 1 feet 11 inches
Multiplication by 2: 11 feet 4 inches

                zero++;

        printf("Matrix is not sparse");

        for (j = 0; j < c; j++)
            if (a[i][j] != 0) {
                printf("%d %d %d\n", i, j, a[i][j]);
                count++;
            }

}

Q1. Write a program to implement Stack using an array, showing overflow and underflow in push and pop operations.
Answer
#include <stdio.h>
#define MAX 5

int stack[MAX], top = -1;

void push(int x) {
    if (top == MAX - 1)
        printf("Stack Overflow!\n");
    else {
        stack[++top] = x;
        printf("%d pushed\n", x);
    }
}

void pop() {
    if (top == -1)
        printf("Stack Underflow!\n");
    else
        printf("Popped: %d\n", stack[top--]);
}

void display() {
    int i;
    for (i = top; i >= 0; i--)
        printf("%d ", stack[i]);
    printf("\n");
}

int main() {
    push(10);
    push(20);
    push(30);
    display();
    pop();
    display();
    return 0;
}

Output
10 pushed
20 pushed
30 pushed
30 20 10
Popped: 30
20 10

Q3. Write a program to implement Circular Queue using an array, showing insertion and deletion operations.
Answer
#include <stdio.h>
#define MAX 5

int q[MAX], front = -1, rear = -1;

void insert(int x) {
    if ((rear + 1) % MAX == front) {
        printf("Queue Overflow!\n");
        return;
    }

    if (front == -1)
        front = 0;

    rear = (rear + 1) % MAX;
    q[rear] = x;
    printf("%d inserted\n", x);
}

void delete() {
    if (front == -1) {
        printf("Queue Underflow!\n");
        return;
    }

    printf("Deleted: %d\n", q[front]);

    if (front == rear)
        front = rear = -1;
    else
        front = (front + 1) % MAX;
}

void display() {
    int i;

    if (front == -1) {
        printf("Queue is empty\n");
        return;
    }

    i = front;
    while (1) {
        printf("%d ", q[i]);
        if (i == rear)
            break;
        i = (i + 1) % MAX;
    }
    printf("\n");
}

int main() {
    insert(10);
    insert(20);
    insert(30);
    display();
    delete();
    insert(40);
    display();
    return 0;
}

WEEK 5: LINKED LIST
Q1. Write a program to implement a Linear Linked List, showing creation, display, insertion, deletion, and searching operations.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

void insert(int x, int pos) {
    struct Node *n = malloc(sizeof(struct Node));
    struct Node *t = head;
    int i;

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = x;

    if (pos == 1) {
        n->next = head;
        head = n;
        return;
    }

    for (i = 1; t && i < pos - 1; i++)
        t = t->next;

    if (t == NULL) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->next = t->next;
    t->next = n;
}

void deleteNode(int pos) {
    struct Node *t = head, *d;
    int i;

    if (pos < 1 || head == NULL) {
        printf("Invalid deletion\n");
        return;
    }

    if (pos == 1) {
        head = head->next;
        free(t);
        return;
    }

    for (i = 1; t && i < pos - 1; i++)
        t = t->next;

    if (t == NULL || t->next == NULL) {
        printf("Invalid position\n");
        return;
    }

    d = t->next;
    t->next = d->next;
    free(d);
}

void display() {
    struct Node *t = head;

    while (t) {
        printf("%d -> ", t->data);
        t = t->next;
    }
    printf("NULL\n");
}

void search(int x) {
    struct Node *t = head;
    int pos = 1;

    while (t) {
        if (t->data == x) {
            printf("Found at position %d\n", pos);
            return;
        }
        t = t->next;
        pos++;
    }
    printf("Element not found\n");
}

int main() {
    insert(10, 1);
    insert(20, 2);
    insert(30, 3);

    printf("Original list: ");
    display();

    insert(15, 2);
    printf("After insertion: ");
    display();

    deleteNode(3);
    printf("After deletion: ");
    display();

    search(30);
    return 0;
}

Output
Original list: 10 -> 20 -> 30 -> NULL
After insertion: 10 -> 15 -> 20 -> 30 -> NULL
After deletion: 10 -> 15 -> 30 -> NULL
Found at position 3

Q1. Write a program to display the reverse of a linked list.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

void insert(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    struct Node *t;

    if (n == NULL)
        return;

    n->data = x;
    n->next = NULL;

    if (head == NULL) {
        head = n;
        return;
    }

    t = head;
    while (t->next)
        t = t->next;

    t->next = n;
}

void reverse() {
    struct Node *prev = NULL, *cur = head, *next;

    while (cur) {
        next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }

    head = prev;
}

void display() {
    struct Node *t = head;

    while (t) {
        printf("%d -> ", t->data);
        t = t->next;
    }
    printf("NULL\n");
}

int main() {
    insert(10);
    insert(20);
    insert(30);
    insert(40);

    printf("Original list: ");
    display();

    reverse();

    printf("Reversed list: ");
    display();

    return 0;
}

Output
Original list: 10 -> 20 -> 30 -> 40 -> NULL
Reversed list: 40 -> 30 -> 20 -> 10 -> NULL

Q3. Write a program to implement a Doubly Linked List, showing creation, display, insertion, deletion, and searching operations.
Answer
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev, *next;
};

struct Node *head = NULL;

void insert(int x, int pos) {
    struct Node *n, *t;
    int i;

    if (pos < 1) {
        printf("Invalid position\n");
        return;
    }

    n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->data = x;
    n->prev = n->next = NULL;

    if (pos == 1) {
        n->next = head;

        if (head)
            head->prev = n;

        head = n;
        return;
    }

    t = head;

    for (i = 1; t && i < pos - 1; i++)
        t = t->next;

    if (t == NULL) {
        printf("Invalid position\n");
        free(n);
        return;
    }

    n->next = t->next;
    n->prev = t;

    if (t->next)
        t->next->prev = n;

    t->next = n;
}

void deleteNode(int pos) {
    struct Node *t = head;
    int i;

    if (pos < 1 || head == NULL) {
        printf("Invalid deletion\n");
        return;
    }

    for (i = 1; t && i < pos; i++)
        t = t->next;

    if (t == NULL) {
        printf("Invalid position\n");
        return;
    }

    if (t->prev)
        t->prev->next = t->next;
    else
        head = t->next;

    if (t->next)
        t->next->prev = t->prev;

    free(t);
}

void display() {
    struct Node *t = head;

    while (t) {
        printf("%d <-> ", t->data);
        t = t->next;
    }
    printf("NULL\n");
}

void search(int x) {
    struct Node *t = head;
    int pos = 1;

    while (t) {
        if (t->data == x) {
            printf("Found at position %d\n", pos);
            return;
        }

        t = t->next;
        pos++;
    }

    printf("Element not found\n");
}

    insert(10, 1);
    insert(20, 2);
    insert(30, 3);

    printf("Original list: ");
    display();

    insert(15, 2);
    printf("After insertion: ");
    display();

    deleteNode(3);
    printf("After deletion: ");
    display();

    search(30);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *n = malloc(sizeof(struct Node));

Q2. Write a program to sort a given array using Quick Sort.
Answer
#include <stdio.h>

void quickSort(int a[], int low, int high) {
    int i, j, pivot, temp;

    if (low >= high)
        return;

    pivot = a[low];
    i = low + 1;
    j = high;

    while (i <= j) {
        while (i <= high && a[i] <= pivot)
            i++;

        while (j > low && a[j] > pivot)
            j--;

        if (i < j) {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    a[low] = a[j];
    a[j] = pivot;

    quickSort(a, low, j - 1);
    quickSort(a, j + 1, high);
}

void merge(int a[], int low, int mid, int high) {
    int i = low, j = mid + 1, k = low;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= high)
        temp[k++] = a[j++];

    for (i = low; i <= high; i++)
        a[i] = temp[i];
}

    }
}

Q3. Write a program to implement Prim's algorithm for a Minimum Spanning Tree (MST).
Answer
#include <stdio.h>

#define MAX 20
#define INF 999999

int main() {
    int graph[MAX][MAX], selected[MAX] = {0};
    int n, i, j, min, x, y;
    int edges = 0, cost = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    if (n < 1 || n > MAX) {
        printf("Invalid number of vertices");
        return 0;
    }

    printf("Enter weighted adjacency matrix:\n");

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);

    selected[0] = 1;

    printf("Edges in MST:\n");

    while (edges < n - 1) {
        min = INF;
        x = -1;
        y = -1;

        for (i = 0; i < n; i++) {
            if (selected[i]) {
                for (j = 0; j < n; j++) {
                    if (!selected[j] &&
                        graph[i][j] != 0 &&
                        graph[i][j] < min) {
                        min = graph[i][j];
                        x = i;
                        y = j;
                    }
                }
            }
        }

        if (x == -1) {
            printf("Graph is disconnected; MST does not exist.\n");
            return 0;
        }

        printf("%d - %d : Weight %d\n", x, y, min);

        cost += min;
        selected[y] = 1;
        edges++;
    }

    printf("Total cost of MST = %d\n", cost);

    return 0;
}
